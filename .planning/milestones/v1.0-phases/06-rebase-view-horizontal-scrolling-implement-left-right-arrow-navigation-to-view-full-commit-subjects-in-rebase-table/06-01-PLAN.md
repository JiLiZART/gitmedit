**3. Replace scroll offset calculation (lines 319-328) with terminal-line-aware version:**

The new scroll offset must count terminal lines consumed by rows, not row count. Strategy:
- Sum terminal lines for all rows before the selected row
- Target: place selected row near vertical center
- Find `scroll_offset_rows` (number of rows to skip) such that accumulated terminal lines of skipped rows >= `lines_before_selected - visible_height/2`

```rust
let visible_height = area.height as usize;
let scroll_offset = if let Some(idx) = selected_line_idx {
    let lines_before: usize = row_heights[..idx].iter().sum();
    let half_visible = visible_height / 2;
    if lines_before >= half_visible {
        let target_skip = lines_before.saturating_sub(half_visible);
        let mut accumulated = 0usize;
        let mut skip_rows = 0;
        for &h in &row_heights {
            if accumulated >= target_skip {
                break;
            }
            accumulated += h;
            skip_rows += 1;
        }
        skip_rows
    } else {
        0
    }
} else {
    0
};
```

**4. Replace `.take(visible_height)` with terminal-line-aware take:**

Instead of `.take(visible_height)` which counts rows, accumulate terminal lines and stop when the visible area is full:

```rust
let mut lines_used = 0usize;
let rows: Vec<Row> = rebase_lines
    .iter()
    .zip(row_heights.iter())
    .enumerate()
    .skip(scroll_offset)
    .take_while(|(_, (_, &h))| {
        // IMPORTANT: renamed from h to height ref
        let fits = lines_used < visible_height;
        // We don't add to lines_used here since take_while checks BEFORE consuming
        fits
    })
    .map(|(i, (line, &h))| {
        lines_used += h;
        // ... existing match logic ...
    })
    .collect();
```

Note: The `take_while` + `lines_used` pattern needs careful handling. A cleaner approach is to collect into a temporary vec with a manual loop:

```rust
let mut rows: Vec<Row> = Vec::new();
let mut lines_used = 0usize;
for (i, (line, &h)) in rebase_lines.iter().zip(row_heights.iter()).enumerate().skip(scroll_offset) {
    if lines_used + h > visible_height && !rows.is_empty() {
        break;
    }
    lines_used += h;
    let is_selected = selected_line_idx == Some(i);
    let row = match line {
        RebaseLine::Action { action, hash, subject } => {
            let wrapped_lines = wrap_subject(subject, wrap_width);
            let height = wrapped_lines.len() as u16;
            let subject_text = wrapped_lines.join("\n");
            let action_style = match action {
                RebaseAction::Pick   => Style::default().fg(Color::Green),
                RebaseAction::Squash => Style::default().fg(Color::Yellow),
                RebaseAction::Fixup  => Style::default().fg(Color::Cyan),
                RebaseAction::Drop   => Style::default().fg(Color::Red),
                RebaseAction::Exec   => Style::default().fg(Color::Magenta),
            };
            let row_style = if is_selected {
                Style::default().bg(Color::DarkGray).add_modifier(Modifier::BOLD)
            } else {
                Style::default()
            };
            Row::new(vec![
                Cell::from(format!("{:<7}", action.as_str())).style(action_style),
                Cell::from(hash.chars().take(7).collect::<String>()),
                Cell::from(subject_text),
            ])
            .height(height)
            .style(row_style)
        }
        RebaseLine::Comment(text) => {
            Row::new(vec![
                Cell::from(text.clone()),
            ])
            .style(Style::default().fg(Color::DarkGray))
        }
    };
    rows.push(row);
}
```
