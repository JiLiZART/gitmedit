**In src/document.rs:**

1. Add function `detect_squash_header(raw: &str, comment_char: char) -> Option<usize>`:
   - Scan lines for one that starts with `"{comment_char} This is a combination of"` (the git-generated squash marker).
   - If found, continue scanning forward. The header includes all consecutive lines starting with comment_char from that point.
   - Return `Some(header_end_index)` where header_end_index is the line index of the first non-comment line after the squash header block.
   - If no squash marker found, return None.
   - Note: The header detection must use comment_char (not hardcoded '#') because git respects core.commentChar.

The function signature:
```rust
pub fn detect_squash_header(raw: &str, comment_char: char) -> Option<usize> {
    let lines: Vec<&str> = raw.split('\n').collect();
    // Skip trailing empty token from split
    let lines = if raw.ends_with('\n') && lines.len() > 1 {
        &lines[..lines.len() - 1]
    } else {
        &lines[..]
    };

    let marker = format!("{} This is a combination of", comment_char);
    let mut in_header = false;

    for (i, line) in lines.iter().enumerate() {
        if !in_header && line.starts_with(&marker) {
            in_header = true;
            continue;
        }
        if in_header && !line.starts_with(comment_char) {
            return Some(i);
        }
    }

    // If we entered header but never found a non-comment line,
    // the entire file is header (edge case)
    if in_header {
        return Some(lines.len());
    }

    None
}
```

pub fn render(frame: &mut Frame, app: &App) {
    let chunks = Layout::default()
        .direction(Direction::Vertical)
        .constraints([Constraint::Min(0), Constraint::Length(1)])
        .split(frame.area());

    match app.context() {
        GitContext::Rebase => {
            Self::render_rebase_table(frame, app, chunks[0]);
            Self::render_rebase_status_bar(frame, app, chunks[1]);

            Self::render_squash_mode(frame, app, chunks[0]);
            Self::render_squash_status_bar(frame, app, chunks[1]);
        }
        _ => {
            Self::render_content(frame, app, chunks[0]);
            Self::render_status_bar(frame, app, chunks[1]);
        }
    }

    if app.is_help_visible() {

    }
}

fn render_squash_mode(frame: &mut Frame, app: &App, area: Rect) {
    let log_lines = app.squash_log();
    // Log section: fixed height based on number of log lines, capped at 40% of area
    let log_height = (log_lines.len() as u16 + 2).min(area.height * 40 / 100); // +2 for border

    let layout = Layout::default()
        .direction(Direction::Vertical)
        .constraints([
            Constraint::Length(log_height),
            Constraint::Min(5),
        ])
        .split(area);

    let log_area = layout[0];
    let msg_area = layout[1];

    // Render read-only commit log
    let log_text: Vec<Line> = log_lines.iter().map(|line| {
        Line::from(Span::styled(
            line.clone(),
            Style::default().fg(Color::DarkGray),
        ))
    }).collect();

    let log_widget = Paragraph::new(log_text)
        .block(Block::default()
            .title(" Commit Log (read-only) ")
            .borders(Borders::ALL)
            .style(Style::default().bg(Color::Black)))
        .style(Style::default().fg(Color::DarkGray));

    frame.render_widget(log_widget, log_area);

    // Render editable message using the same render_content logic
    // but scoped to the message area
    Self::render_content(frame, app, msg_area);
}
```

fn render_squash_status_bar(frame: &mut Frame, app: &App, area: Rect) {
    // Reuse commit status bar logic (subject counter, blank line warning)
    // but add squash context indicator
    let first_line = app.document().first_line();
    let char_count = first_line.chars().count();
    let counter_color = counter_color_for(char_count);
    let counter_text = format!("Chars: {}", char_count);
    let counter_span = Span::styled(
        counter_text,
        Style::default().fg(counter_color).add_modifier(Modifier::BOLD),
    );

    let has_blank = app.document().has_blank_line_after_subject();
    let blank_warning = blank_warning_span(has_blank);

    let actions_span = Span::raw("  |  ^S Save  Esc Cancel  ^H Help  [Squash]");

    let status_line = Line::from(vec![counter_span, blank_warning, actions_span]);
    let status_widget = Paragraph::new(status_line)
        .style(Style::default().fg(Color::White).bg(Color::DarkGray));
    frame.render_widget(status_widget, area);
}
```
